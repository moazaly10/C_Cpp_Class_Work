# Makefile.txt — build objects, link executable, clean

CC      := gcc
CFLAGS  := -std=c11 -Wall -Wextra -O2

TARGET  := use_linked_list
OBJS    := linked_list.o use_linked_list.o
HEADERS := linked_list.h

.PHONY: all objects link clean run

# build everything
all: objects link

# object files
objects: $(OBJS)

# link executable
link: $(TARGET)

$(TARGET): $(OBJS)
	$(CC) $(CFLAGS) -o $@ $(OBJS)

# compile rule
%.o: %.c $(HEADERS)
	$(CC) $(CFLAGS) -c $< -o $@

# remove build outputs
clean:
	$(RM) -f $(OBJS) $(TARGET)

# run program
run: $(TARGET)
	./$(TARGET)
